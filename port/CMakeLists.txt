cmake_minimum_required(VERSION 3.16)
project(sakurasou C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE RelWithDebInfo)
endif()
option(SAKURA_CONSOLE "Windows: build with a console window (developer modes and logs)" OFF)
option(SAKURA_STATIC_DEPS "Link SDL2 and FFmpeg statically (release builds; see tools/build_linux.sh)" OFF)
find_package(SDL2 REQUIRED)
find_package(PkgConfig REQUIRED)
if(SAKURA_STATIC_DEPS)
  pkg_check_modules(AV REQUIRED libavcodec libavformat libavutil libswscale libswresample)
else()
  pkg_check_modules(AV REQUIRED IMPORTED_TARGET libavcodec libavformat libavutil libswscale libswresample)
endif()
add_library(clhca STATIC third_party/clhca/clhca.c)
target_include_directories(clhca PUBLIC third_party/clhca)
file(GLOB SRC CONFIGURE_DEPENDS src/*.cpp)
list(FILTER SRC EXCLUDE REGEX " \\([0-9]+\\)\\.cpp$")   # stray copies like "adv (2).cpp"
add_executable(sakurasou ${SRC})
if(WIN32 AND TARGET SDL2::SDL2main)
  target_link_libraries(sakurasou PRIVATE SDL2::SDL2main)   # must precede SDL2 on the link line
endif()
if(SAKURA_STATIC_DEPS)
  target_include_directories(sakurasou PRIVATE ${AV_STATIC_INCLUDE_DIRS})
  target_link_directories(sakurasou PRIVATE ${AV_STATIC_LIBRARY_DIRS})
  target_link_libraries(sakurasou PRIVATE SDL2::SDL2-static clhca ${AV_STATIC_LIBRARIES})
  if(UNIX AND NOT APPLE)
    target_link_options(sakurasou PRIVATE -static-libgcc -static-libstdc++)
  endif()
else()
  target_link_libraries(sakurasou PRIVATE SDL2::SDL2 clhca PkgConfig::AV)
endif()
target_compile_options(sakurasou PRIVATE -Wall -Wno-unused-variable -Wno-unused-function)
if(WIN32)
  # 64-bit file offsets for the 1.5 GB ISO, UTF-8 as the process code page (fopen with UTF-8 paths).
  target_compile_definitions(sakurasou PRIVATE _FILE_OFFSET_BITS=64)
  target_sources(sakurasou PRIVATE win/sakurasou.rc)
  if(NOT SAKURA_CONSOLE)
    set_target_properties(sakurasou PROPERTIES WIN32_EXECUTABLE ON)
  endif()
  if(MINGW)
    target_link_options(sakurasou PRIVATE -static-libgcc -static-libstdc++)
  endif()
endif()
