cmake_minimum_required(VERSION 3.16)
project(sakurasou C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE RelWithDebInfo)
endif()
find_package(SDL2 REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(AV REQUIRED IMPORTED_TARGET libavcodec libavformat libavutil libswscale libswresample)
add_library(clhca STATIC third_party/clhca/clhca.c)
target_include_directories(clhca PUBLIC third_party/clhca)
file(GLOB SRC CONFIGURE_DEPENDS src/*.cpp)
add_executable(sakurasou ${SRC})
target_link_libraries(sakurasou PRIVATE SDL2::SDL2 clhca PkgConfig::AV)
target_compile_options(sakurasou PRIVATE -Wall -Wno-unused-variable -Wno-unused-function)
